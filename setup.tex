% Panel layout support
\usepackage{subcaption}
\usepackage{cleveref}
\renewcommand{\thesubfigure}{\Alph{subfigure}}
\captionsetup[subfigure]{labelformat=simple,labelsep=none}

% Bold sans-serif panel label overlaid top-left of each panel image.
\newcommand{\panel}[2]{%
  \begin{tikzpicture}
    \node[anchor=north west,inner sep=0pt] (panelimg) {#2};
    \node[anchor=north west,inner sep=1.5pt,
          font=\sffamily\bfseries\large] at (panelimg.north west) {#1};
  \end{tikzpicture}%
}

% Include a panel PDF if it exists; otherwise draw a gray placeholder box so the
% document still compiles (e.g. on Overleaf before panels have been generated).
\newcommand{\panelgraphic}[2][]{%
  \IfFileExists{#2}{\includegraphics[#1]{#2}}{%
    \fbox{\parbox[c][0.6\linewidth][c]{0.95\linewidth}{\centering\small\ttfamily
      \detokenize{#2}\missing}}}%
}
