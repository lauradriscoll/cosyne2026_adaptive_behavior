% Panel layout support
\usepackage{subcaption}
\usepackage{cleveref}
\renewcommand{\thesubfigure}{\Alph{subfigure}}
\captionsetup[subfigure]{labelformat=simple,labelsep=none}

% Bold sans-serif panel label overlaid top-left of each panel image.
\newcommand{\panel}[2]{%
  \begin{tikzpicture}
    \node[anchor=north west,inner sep=0pt] (panelimg) {#2};
    \node[anchor=north west,inner sep=1.5pt,
          font=\sffamily\bfseries\large] at (panelimg.north west) {#1};
  \end{tikzpicture}%
}
